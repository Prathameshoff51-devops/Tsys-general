ALTER PROCEDURE [dbo].[spSetupDataForDiscoverMerchantFile]
AS

BEGIN

	DECLARE @CurrentRunDate DATETIME = GETUTCDATE()
	DECLARE @SubmittingStatusId INT = 1 -- Submitting StatusId
				
	IF OBJECT_ID('tempdb..#tempData') IS NOT NULL DROP TABLE #tempData

	SELECT
		dbm.DiscoverMerchantId
	INTO #tempData
	FROM [dbo].[vuDiscoverBoardingMerchant] dbm
	JOIN [dbo].[DiscoverMerchantBoardingChange] dbc ON dbc.DiscoverMerchantId = dbm.DiscoverMerchantId
	WHERE dbc.IsChanged = 1 AND ISNULL(dbm.SubmissionStatusId, '') <> @SubmittingStatusId 	

	BEGIN
		BEGIN TRY	
			BEGIN TRANSACTION			
						
			-- UPDATE CHANGES FIRST	
			UPDATE s
				SET [SubmissionStatusId] = @SubmittingStatusId,					
					[SentDateUTC] = @CurrentRunDate,
					[ReceivedDateUTC] = NULL
			FROM [dbo].[DiscoverMerchantSubmission] s
				JOIN #tempData t ON s.[DiscoverMerchantId] = t.[DiscoverMerchantId]
			WHERE s.[DiscoverMerchantId] IS NOT NULL

			-- INSERT NEW RECORDS LAST
			INSERT INTO [dbo].[DiscoverMerchantSubmission]
			   ([DiscoverMerchantId]
			   ,[SubmissionStatusId]			
			   ,[SentDateUTC]
			   ,[ReceivedDateUTC])
			SELECT 
				t.DiscoverMerchantId
				, @SubmittingStatusId			
				, @CurrentRunDate
				, NULL
			FROM #tempData t
			LEFT JOIN [dbo].[DiscoverMerchantSubmission] s WITH (NOLOCK) ON s.[DiscoverMerchantId] = t.[DiscoverMerchantId]
			WHERE s.[DiscoverMerchantId] IS NULL
				
			COMMIT TRANSACTION	
		END TRY
		BEGIN CATCH
			IF @@TRANCOUNT > 0
				ROLLBACK TRANSACTION

			DECLARE @ErrorMessage NVARCHAR(4000)
				, @ErrorSeverity INT
				, @ErrorState INT

			SELECT @ErrorMessage = ERROR_MESSAGE()
				, @ErrorSeverity = ERROR_SEVERITY()
				, @ErrorState = ERROR_STATE()

			RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState)
		END CATCH
	END

	-- Drop Temp Tables
	DROP TABLE #tempData	
END
