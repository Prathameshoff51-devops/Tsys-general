ALTER   PROCEDURE [dbo].[spPopulateDiscoverBoardingMerchantsForFile]
AS

BEGIN
	-- Truncate existing records
	TRUNCATE TABLE dbo.DiscoverBoardingMerchant

	BEGIN TRY	
		BEGIN TRANSACTION			
						
		INSERT INTO dbo.DiscoverBoardingMerchant (
			DiscoverMerchantID,
			PayPalMerchantID,
			IsHPS,	
			SequenceKey,	
			ProcessorOID,
			DiscoverSENumber,
			PayPalSE,
			AcquiringProgram,
			MerchantName,
			DBAAddress1,
			DBAAddress2,	
			DBACity,
			DBAState,
			DBAZip,
			PhoneNbr,	
			FirstName,
			LastName,	
			BusinessPrincipalHomeAddress1, 
			BusinessPrincipalHomeAddress2, 
			BusinessPrincipalHomeCity, 
			BusinessPrincipalHomeState, 
			BusinessPrincipalHomeZip,
			FedTaxID,
			LegalName,
			LegalAddress1,
			LegalAddress2,
			LegalCity,
			LegalState,
			LegalZip,
			SicCode,
			DiscoverBusinessType,	
			DiscoverOptOutReasonID,
			OptOutReasonCategory,		
			DiscoverCode,	
			DiscoverStatusID,
			DiscoverBatchActionID,	
			SubmissionStatusId,
			SentToDiscv,
			OwnerSSNEncrypted,
			MerchantStatus,
			StatusDate,
			DiscoverMerchantRecordCreatedDate,
			IsRetainedMcc,
			SubmissionTypeId,
			AcqBankID,
			AcquirerId,
			DebtRepaymentIndicator)
		SELECT TOP 100000
			DiscoverMerchantID,
			PayPalMerchantID,
			IsHPS,	
			SequenceKey,	
			ProcessorOID,
			DiscoverSENumber,
			PayPalSE,
			AcquiringProgram,
			MerchantName,
			REPLACE(REPLACE(DBAAddress1, CHAR(13), ' '), CHAR(10), ' '),
			REPLACE(REPLACE(DBAAddress2, CHAR(13), ' '), CHAR(10), ' '),
			DBACity,
			DBAState,
			DBAZip,
			PhoneNbr,	
			FirstName,
			LastName,	
			REPLACE(REPLACE(BusinessPrincipalHomeAddress1, CHAR(13), ' '), CHAR(10), ' '),
			REPLACE(REPLACE(BusinessPrincipalHomeAddress2, CHAR(13), ' '), CHAR(10), ' '),
			BusinessPrincipalHomeCity, 
			BusinessPrincipalHomeState, 
			BusinessPrincipalHomeZip,
			FedTaxID,
			LegalName,
			LegalAddress1,
			LegalAddress2,
			LegalCity,
			LegalState,
			LegalZip,
			SicCode,
			DiscoverBusinessType,	
			DiscoverOptOutReasonID,
			OptOutReasonCategory,		
			DiscoverCode,	
			DiscoverStatusID,
			DiscoverBatchActionID,	
			SubmissionStatusId,
			SentToDiscv,
			OwnerSSNEncrypted,
			MerchantStatus,
			StatusDate,
			DiscoverMerchantRecordCreatedDate,
			IsRetainedMcc,
			SubmissionTypeId,
			AcqBankID,
			AcquirerId,
			DebtRepaymentIndicator
		FROM [dbo].[vuDiscoverBoardingMerchant] WITH (NOLOCK)
			WHERE SubmissionStatusId = 1 -- Submitting
		ORDER BY DiscoverMerchantID DESC
				
		COMMIT TRANSACTION	
	END TRY
	BEGIN CATCH
		IF @@TRANCOUNT > 0
			ROLLBACK TRANSACTION

		DECLARE @ErrorMessage NVARCHAR(4000)
			, @ErrorSeverity INT
			, @ErrorState INT

		SELECT @ErrorMessage = ERROR_MESSAGE()
			, @ErrorSeverity = ERROR_SEVERITY()
			, @ErrorState = ERROR_STATE()

		RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState)
	END CATCH
END
